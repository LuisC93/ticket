const SHEET_ID = '1T9z7zVtb4CV-uH8uP5EL54FwyWeie8GKoxP0yAQkMhY';

const SHEETS = {
  monitoreo: 'Loocker Studio',
  base: 'base general',
  sla: 'SLA',
  slacfo: 'CFO SLA GENERAL'
};

// Mapeo tipo incidencia → estado monitor
const ESTADO_MAP = {
  'Corte F.O':           'Corte F.O externa',
  'Corte F.O externa':   'Corte F.O externa',
  'Intervenida':         'Intervenida',
  'Falla de equipo':     'Intervenida',
  'Sin señal':           'Corte F.O externa',
  'Gabinete apagado':    'Intervenida',
  'RT Apagado':          'Intervenida',
  'SW OFF':              'Intervenida',
  'AP OFF':              'Intervenida',
  'FW Apagado':          'Intervenida',
  'sin conexion':        'Corte F.O externa',
  'Sin conexion':        'Corte F.O externa'
};

function doGet(e) {
  try {
    const ss      = SpreadsheetApp.openById(SHEET_ID);
    const action  = e.parameter.action  || '';
    const sheet   = e.parameter.sheet   || 'SLA';
    const payload = e.parameter.payload || '';

    console.log('📥 APPS SCRIPT RECIBIÓ:', { action, sheet, hasPayload: !!payload });

    // ── Escrituras via payload ──
    if (payload) {
      try {
        const data = JSON.parse(decodeURIComponent(payload));
        console.log('📦 Payload decodificado:', data.action);
        
        if      (data.action === 'append')        return output(handleAppend(ss, data));
        else if (data.action === 'update')        return output(handleUpdate(ss, data));
        else if (data.action === 'updateEstado')  {
          // updateEstado es NO-BLOQUEANTE: lo procesamos async
          handleUpdateEstadoAsync(ss, data);
          return output({ status: 'ok', msg: 'updateEstado en progreso' });
        }
        else return output({ status: 'error', msg: 'Accion desconocida: ' + data.action });
      } catch(parseErr) {
        console.error('❌ Error al parsear payload:', parseErr);
        return output({ status: 'error', msg: 'Error al parsear payload: ' + parseErr.message });
      }
    }

    // ── Lecturas via action ──
    if (action === 'ping')         return output({ status: 'ok', msg: 'Conectado', timestamp: new Date().toISOString() });
    if (action === 'getAll')       return output(handleGetAll(ss, sheet));
    if (action === 'getBloques')   return output(handleGetBloques(ss));
    if (action === 'updateEstado') return output(handleUpdateEstado(ss, e.parameter));

    // Sin action → devuelve todas las hojas
    const result = {};
    for (let key in SHEETS) {
      try {
        result[key] = getSheetData(ss, SHEETS[key]);
      } catch(err) {
        result[key] = [];
      }
    }
    return output(result);

  } catch (err) {
    console.error('❌ ERROR GENERAL:', err);
    return output({ error: err.message, stack: err.stack });
  }
}

// ── GET BLOQUES ──
function handleGetBloques(ss) {
  try {
    const sheet = ss.getSheetByName('base general');
    if (!sheet) return { status: 'error', msg: 'Hoja base general no encontrada' };
    
    const lastRow = sheet.getLastRow();
    if (lastRow < 2) return { status: 'ok', map: {} };
    
    const data = sheet.getRange(2, 1, lastRow - 1, 9).getValues();
    const map = {};
    
    data.forEach(function(row) {
      const cod    = String(row[0]).trim();
      const bloque = String(row[8]).trim();
      if (cod && bloque && bloque !== '') map[cod] = bloque;
    });
    
    console.log('✓ getBloques:', Object.keys(map).length + ' códigos cargados');
    return { status: 'ok', map: map };
  } catch(err) {
    console.error('❌ handleGetBloques error:', err);
    return { status: 'error', msg: err.message };
  }
}

// ── GET ALL ──
function handleGetAll(ss, sheetName) {
  try {
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) return { status: 'error', msg: 'Hoja no encontrada: ' + sheetName };
    
    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) return { status: 'ok', rows: [] };
    
    const rows = sheet.getRange(2, 1, lastRow - 1, 14).getValues();
    console.log('✓ getAll:', rows.length + ' registros de ' + sheetName);
    return { status: 'ok', rows: rows };
  } catch(err) {
    console.error('❌ handleGetAll error:', err);
    return { status: 'error', msg: err.message };
  }
}

// ── APPEND (Crear nuevo ticket) ──
function handleAppend(ss, data) {
  try {
    const sheet = ss.getSheetByName(data.sheet || 'SLA');
    if (!sheet) return { status: 'error', msg: 'Hoja no encontrada: ' + (data.sheet || 'SLA') };
    
    const row = data.row || [];
    if (row.length !== 14) {
      console.warn('⚠️ Append: Se esperaban 14 columnas, se recibieron ' + row.length);
    }
    
    sheet.appendRow(row);
    const lastRow = sheet.getLastRow();
    colorearFila(sheet, lastRow, row[13] || 'Abierto');
    
    console.log('✓ Ticket creado:', { fila: lastRow, codigo: row[3], estado: row[13] });
    return { status: 'ok', row: lastRow, msg: 'Ticket guardado' };
  } catch(err) {
    console.error('❌ handleAppend error:', err);
    return { status: 'error', msg: 'Error al guardar: ' + err.message };
  }
}

// ── UPDATE (Actualizar ticket existente) ──
function handleUpdate(ss, data) {
  try {
    const sheet = ss.getSheetByName(data.sheet || 'SLA');
    if (!sheet) return { status: 'error', msg: 'Hoja no encontrada' };
    
    const lastRow = sheet.getLastRow();
    if (lastRow < 2) return { status: 'error', msg: 'Hoja vacía' };
    
    const ticketId = String(data.ticketId);
    const codCol   = sheet.getRange(2, 4, lastRow - 1, 1).getValues();
    let targetRow  = -1;
    
    for (let i = 0; i < codCol.length; i++) {
      if (String(codCol[i][0]) === ticketId) { targetRow = i + 2; break; }
    }
    
    if (targetRow === -1) return { status: 'error', msg: 'COD no encontrado: ' + ticketId };
    
    if (data.horaFinal) sheet.getRange(targetRow, 11).setValue(data.horaFinal);
    if (data.duracion)  sheet.getRange(targetRow, 12).setValue(data.duracion);
    if (data.motivo)    sheet.getRange(targetRow, 9).setValue(data.motivo);
    if (data.estado)    sheet.getRange(targetRow, 14).setValue(data.estado);
    
    colorearFila(sheet, targetRow, data.estado || 'Cerrado');
    
    console.log('✓ Ticket actualizado:', { fila: targetRow, codigo: ticketId, estado: data.estado });
    return { status: 'ok', updatedRow: targetRow, msg: 'Ticket actualizado' };
  } catch(err) {
    console.error('❌ handleUpdate error:', err);
    return { status: 'error', msg: 'Error al actualizar: ' + err.message };
  }
}

// ── UPDATE ESTADO MONITOR (Síncrono - para compatibilidad) ──
function handleUpdateEstado(ss, data) {
  try {
    const result = updateEstadoInternal(ss, data);
    console.log('✓ updateEstado sincrónico:', result);
    return result;
  } catch(err) {
    console.error('❌ handleUpdateEstado error:', err);
    return { status: 'warn', msg: 'No se pudo actualizar estado: ' + err.message };
  }
}

// ── UPDATE ESTADO MONITOR (Asincrónico - no bloquea append) ──
function handleUpdateEstadoAsync(ss, data) {
  try {
    const result = updateEstadoInternal(ss, data);
    console.log('✓ updateEstadoAsync completado:', result);
  } catch(err) {
    console.warn('⚠️ updateEstadoAsync falló (no crítico):', err.message);
  }
}

// ── LÓGICA COMPARTIDA DE UPDATE ESTADO ──
function updateEstadoInternal(ss, data) {
  var nombreHoja = String(data.monitor || '').trim();
  if (!nombreHoja) return { status: 'warn', msg: 'Monitor no especificado' };

  // Buscar la hoja del monitor
  var sheet = ss.getSheetByName(nombreHoja);
  if (!sheet) {
    // Intenta con capitalización
    nombreHoja = nombreHoja.charAt(0).toUpperCase() + nombreHoja.slice(1).toLowerCase();
    sheet = ss.getSheetByName(nombreHoja);
  }
  if (!sheet) {
    console.warn('⚠️ Hoja del monitor no encontrada:', nombreHoja);
    return { status: 'warn', msg: 'Hoja del monitor no encontrada: ' + nombreHoja };
  }

  var tipoInc = String(data.tipoInc || '').trim();
  var estado  = ESTADO_MAP[tipoInc] || 'Navegación estable';
  if (data.cerrar === 'true') estado = 'Navegación estable';

  var fecha = data.fecha ? new Date(data.fecha + 'T00:00:00') : new Date(new Date().getTime() - 6 * 60 * 60 * 1000);
  var diaNum = fecha.getUTCDate(), mesNum = fecha.getUTCMonth();

  var sheetData = sheet.getDataRange().getValues();

  // Buscar columna del día en fila 3 (índice 2)
  var colDia = -1;
  var headerRow = sheetData[2] || [];
  for (var c = 0; c < headerRow.length; c++) {
    var cell = headerRow[c];
    if (cell instanceof Date) {
      if (cell.getDate() === diaNum && cell.getMonth() === mesNum) { colDia = c; break; }
    } else {
      var match = String(cell).match(/(\d{1,2})\/(\d{1,2})/);
      if (match && parseInt(match[1]) === diaNum && parseInt(match[2]) - 1 === mesNum) { colDia = c; break; }
    }
  }
  if (colDia === -1) {
    console.warn('⚠️ Columna del día no encontrada para:', diaNum + '/' + (mesNum + 1));
    return { status: 'warn', msg: 'Columna del día no encontrada' };
  }

  // Buscar fila del código
  var cod = String(data.cod || '').trim();
  var filaEncontrada = -1;
  for (var r = 3; r < sheetData.length; r++) {
    if (String(sheetData[r][0]).trim() === cod) { filaEncontrada = r + 1; break; }
  }
  if (filaEncontrada === -1) {
    console.warn('⚠️ Código no encontrado en hoja:', cod);
    return { status: 'warn', msg: 'Código no encontrado: ' + cod };
  }

  // Escribir estado y color
  var rango = sheet.getRange(filaEncontrada, colDia + 1);
  rango.setValue(estado);
  
  var colores = {
    'Navegación estable': { bg: '#00b050', fg: '#ffffff' },
    'Corte F.O externa':  { bg: '#ff0000', fg: '#ffffff' },
    'Intervenida':        { bg: '#ff0000', fg: '#ffffff' }
  };
  var color = colores[estado] || { bg: '#ffffff', fg: '#000000' };
  rango.setBackground(color.bg);
  rango.setFontColor(color.fg);

  console.log('✓ Estado monitor actualizado:', { hoja: nombreHoja, fila: filaEncontrada, estado: estado });
  return { status: 'ok', hoja: nombreHoja, fila: filaEncontrada, estado: estado };
}

// ── HELPERS ──
function getSheetData(ss, sheetName) {
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) throw new Error('No existe la hoja: ' + sheetName);
  
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  
  const headers = data[0].map(h => String(h).trim());
  return data.slice(1)
    .filter(row => row[0] && String(row[0]).trim() !== '')
    .map(row => {
      const obj = {};
      headers.forEach((h, i) => {
        const val = row[i];
        obj[h] = val instanceof Date ? val.toISOString() : (val !== null ? String(val).trim() : '');
      });
      return obj;
    });
}

function colorearFila(sheet, rowNum, estado) {
  const range = sheet.getRange(rowNum, 1, 1, 14);
  if (estado === 'Cerrado') {
    range.setBackground('#e6f4ea'); 
    range.setFontColor('#1a5c2a');
  } else {
    range.setBackground('#fce8e6'); 
    range.setFontColor('#a50000');
  }
}

function output(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
